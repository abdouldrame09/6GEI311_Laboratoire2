    + __init__(name: str, email: str)

' --- CLASSE TICKET ---
class Ticket {
    + ticketId : int
    + title : str
    + status : Statut
    + priority : Priorite
    + creationDate : datetime
    + updateDate : datetime
    + descriptions : list
    + createdBy : User
    + assignedTo : User
    __
    + __init__(title: str, status: Statut = Statut.OUVERT, priority: Priorite = Priorite.NORMALE)
    + assignTo(user: User) : void
    + updateStatus(status: Statut) : void
    + updateInfo(title: str = None, priority: Priorite = None) : void
    + addDescription(description: DescriptionTicket) : void
    + loadDescriptions() : void
}
